datasource db {
  provider = "postgresql"
}

generator kysely {
  provider = "prisma-kysely"
  output   = "../src/database"
  fileName = "types.ts"
}

generator client {
  provider = "prisma-client-js"
}

enum Role {
  CLIENT
  EXPERT
  ADMIN
}

enum ProjectRole {
  CLIENT_ADMIN
  INTERNAL_TEAM
  EXPERT
}

enum TaskStatus {
  TODO
  IN_PROGRESS
  REVIEW
  DONE
}

enum TaskPriority {
  LOW
  MEDIUM
  HIGH
}

enum MilestoneStatus {
  PENDING
  ACTIVE
  REVIEW
  PAID
}

enum TransactionType {
  DEPOSIT
  ESCROW
  SPENT
  FEE
  REFUND
  PAYMENT_RECEIVED
  WITHDRAWAL
}

model User {
  id          String   @id @default(uuid())
  googleId    String   @unique
  email       String   @unique
  name        String
  avatar      String?
  role        Role     @default(CLIENT)
  currentRole Role     @default(CLIENT)
  balance     Decimal  @default(0.0)
  // Expert profile fields
  title       String?
  bio         String?
  skills      Json?
  rate        String?
  location    String?
  badge       String?
  online      Boolean  @default(false)
  createdAt   DateTime @default(now())
  updatedAt   DateTime @updatedAt

  refreshTokens            RefreshToken[]
  projectMembers           ProjectMember[]
  assignedTasks            Task[]
  transactions             Transaction[]
  conversationParticipants ConversationParticipant[]
  messages                 Message[]
  clientQuickTasks         QuickTask[]               @relation("ClientQuickTasks")
  expertQuickTasks         QuickTask[]               @relation("ExpertQuickTasks")
  proposals                Proposal[]
  bids                     Bid[]
  reviews                  Review[]                  @relation("ReviewsReceived")
  reviewsGiven             Review[]                  @relation("ReviewsGiven")
  sentInvitations          Invitation[]              @relation("InvitationsSent")
  receivedInvitations      Invitation[]              @relation("InvitationsReceived")

  @@map("users")
}

model RefreshToken {
  id        String   @id @default(uuid())
  token     String   @unique
  userId    String
  user      User     @relation(fields: [userId], references: [id], onDelete: Cascade)
  device    String?
  expiresAt DateTime
  createdAt DateTime @default(now())

  @@map("refresh_tokens")
}

model Project {
  id          String   @id @default(uuid())
  title       String
  description String?
  budget      Decimal  @default(0.0)
  spent       Decimal  @default(0.0)
  escrow      Decimal  @default(0.0)
  createdAt   DateTime @default(now())
  updatedAt   DateTime @updatedAt

  members      ProjectMember[]
  tasks        Task[]
  milestones   Milestone[]
  transactions Transaction[]
  quickTasks   QuickTask[]
  invitations  Invitation[]

  @@map("projects")
}

model ProjectMember {
  id        String      @id @default(uuid())
  projectId String
  userId    String
  role      ProjectRole @default(CLIENT_ADMIN)
  status    String      @default("ACTIVE")
  rating    Float?
  createdAt DateTime    @default(now())
  updatedAt DateTime    @updatedAt

  project Project @relation(fields: [projectId], references: [id], onDelete: Cascade)
  user    User    @relation(fields: [userId], references: [id], onDelete: Cascade)

  @@unique([projectId, userId])
  @@map("project_members")
}

model Task {
  id          String       @id @default(uuid())
  projectId   String
  title       String
  status      TaskStatus   @default(TODO)
  priority    TaskPriority @default(MEDIUM)
  assigneeId  String?
  comments    Int          @default(0)
  attachments Int          @default(0)
  milestoneId String?
  bucket      String?
  isOutsource Boolean      @default(false)
  budget      Decimal      @default(0.0)
  quickTaskId String?      @unique
  createdAt   DateTime     @default(now())
  updatedAt   DateTime     @updatedAt

  project   Project    @relation(fields: [projectId], references: [id], onDelete: Cascade)
  assignee  User?      @relation(fields: [assigneeId], references: [id], onDelete: SetNull)
  milestone Milestone? @relation(fields: [milestoneId], references: [id], onDelete: SetNull)
  quickTask QuickTask? @relation(fields: [quickTaskId], references: [id], onDelete: SetNull)

  @@map("tasks")
}

model Milestone {
  id           String          @id @default(uuid())
  projectId    String
  title        String
  amount       Decimal         @default(0.0)
  status       MilestoneStatus @default(PENDING)
  progress     Int             @default(0)
  dueDate      DateTime?
  deliverables Json?
  createdAt    DateTime        @default(now())
  updatedAt    DateTime        @updatedAt

  project Project @relation(fields: [projectId], references: [id], onDelete: Cascade)
  tasks   Task[]
  bids    Bid[]
  invitations Invitation[]

  @@map("milestones")
}

model Transaction {
  id           String          @id @default(uuid())
  userId       String
  date         DateTime        @default(now())
  desc         String
  type         TransactionType
  amount       Decimal
  balanceAfter Decimal
  status       String          @default("Success")
  source       String?
  projectId    String?
  createdAt    DateTime        @default(now())

  user    User     @relation(fields: [userId], references: [id], onDelete: Cascade)
  project Project? @relation(fields: [projectId], references: [id], onDelete: SetNull)

  @@map("transactions")
}

enum ConversationContextType {
  QUICK_TASK
  PROJECT
  DIRECT
}

enum MessageType {
  TEXT
  CODE
  FILE
  SYSTEM
}

enum QuickTaskStatus {
  OPEN
  IN_PROGRESS
  REVIEW
  COMPLETED
}

model Conversation {
  id          String                  @id @default(uuid())
  name        String?
  isGroup     Boolean                 @default(false)
  contextType ConversationContextType @default(DIRECT)
  contextRef  String?
  createdAt   DateTime                @default(now())
  updatedAt   DateTime                @updatedAt

  participants ConversationParticipant[]
  messages     Message[]

  @@map("conversations")
}

model ConversationParticipant {
  id             String   @id @default(uuid())
  conversationId String
  userId         String
  joinedAt       DateTime @default(now())

  conversation Conversation @relation(fields: [conversationId], references: [id], onDelete: Cascade)
  user         User         @relation(fields: [userId], references: [id], onDelete: Cascade)

  @@unique([conversationId, userId])
  @@map("conversation_participants")
}

model Message {
  id             String      @id @default(uuid())
  conversationId String
  senderId       String?
  type           MessageType @default(TEXT)
  content        String
  metadata       Json?
  createdAt      DateTime    @default(now())

  conversation Conversation @relation(fields: [conversationId], references: [id], onDelete: Cascade)
  sender       User?        @relation(fields: [senderId], references: [id], onDelete: SetNull)

  @@map("messages")
}

model QuickTask {
  id             String          @id @default(uuid())
  clientId       String
  expertId       String?
  title          String
  description    String
  status         QuickTaskStatus @default(OPEN)
  budget         Decimal         @default(0.0)
  deadline       DateTime?
  proposalsCount Int             @default(0)
  projectId      String?
  createdAt      DateTime        @default(now())
  updatedAt      DateTime        @updatedAt

  client  User     @relation("ClientQuickTasks", fields: [clientId], references: [id], onDelete: Cascade)
  expert  User?    @relation("ExpertQuickTasks", fields: [expertId], references: [id], onDelete: SetNull)
  project Project? @relation(fields: [projectId], references: [id], onDelete: SetNull)

  proposals Proposal[]
  task      Task?

  @@map("quick_tasks")
}

enum ProposalStatus {
  PENDING
  ACCEPTED
  REJECTED
  WITHDRAWN
}

model Proposal {
  id            String         @id @default(uuid())
  quickTaskId   String
  expertId      String
  coverLetter   String
  proposedPrice Decimal
  estimatedDays Int
  status        ProposalStatus @default(PENDING)
  createdAt     DateTime       @default(now())
  updatedAt     DateTime       @updatedAt

  quickTask QuickTask @relation(fields: [quickTaskId], references: [id], onDelete: Cascade)
  expert    User      @relation(fields: [expertId], references: [id], onDelete: Cascade)

  @@map("proposals")
}

model Bid {
  id            String         @id @default(uuid())
  milestoneId   String
  expertId      String
  coverLetter   String
  amount        Decimal
  contractType  String?
  status        ProposalStatus @default(PENDING)
  createdAt     DateTime       @default(now())
  updatedAt     DateTime       @updatedAt

  milestone Milestone @relation(fields: [milestoneId], references: [id], onDelete: Cascade)
  expert    User      @relation(fields: [expertId], references: [id], onDelete: Cascade)

  @@map("bids")
}

model Review {
  id         String   @id @default(uuid())
  reviewerId String
  expertId   String
  rating     Int
  feedback   String?
  taskTitle  String?
  createdAt  DateTime @default(now())

  reviewer User @relation("ReviewsGiven", fields: [reviewerId], references: [id], onDelete: Cascade)
  expert   User @relation("ReviewsReceived", fields: [expertId], references: [id], onDelete: Cascade)

  @@map("reviews")
}

enum InvitationStatus {
  PENDING
  ACCEPTED
  REJECTED
  CANCELLED
}

model Invitation {
  id        String           @id @default(uuid())
  clientId  String
  expertId  String
  projectId String?
  taskId    String?
  milestoneId String?
  message   String?
  budget    Decimal?
  status    InvitationStatus @default(PENDING)
  createdAt DateTime         @default(now())
  updatedAt DateTime         @updatedAt

  client  User     @relation("InvitationsSent", fields: [clientId], references: [id], onDelete: Cascade)
  expert  User     @relation("InvitationsReceived", fields: [expertId], references: [id], onDelete: Cascade)
  project Project? @relation(fields: [projectId], references: [id], onDelete: SetNull)
  milestone Milestone? @relation(fields: [milestoneId], references: [id], onDelete: SetNull)

  @@map("invitations")
}
